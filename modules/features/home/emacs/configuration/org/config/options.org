  		    :font "JetBrainsMono Nerd Font"
  		    :height 120
  		    :weight 'regular)

  		    :font "Roboto"
  		    :height 120
  		    :weight 'regular)

  		    :font "JetBrainsMono Nerd Font"
  		    :height 120
  		    :weight 'regular)

  		    :slant 'italic)
  (set-face-attribute 'font-lock-keyword-face nil
  		    :slant 'italic)
